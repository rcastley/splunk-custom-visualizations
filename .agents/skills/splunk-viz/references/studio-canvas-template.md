# Native Studio Canvas Adapter

Use this pattern for framework-agnostic JavaScript or TypeScript projects generated by `@splunk/create --mode=dashboard-studio-extension`. Keep visualization-specific calculations and drawing in pure functions; keep the Studio API isolated in the entry-point adapter.

## State and DOM setup

```javascript
import { VisualizationAPI } from '@splunk/dashboard-studio-extension';
import './visualization.css';

const root = document.getElementById('root') || document.body;
const canvas = document.createElement('canvas');
canvas.className = 'my-viz-canvas';
root.replaceChildren(canvas);

const state = {
    dataSources: {},
    loading: false,
    options: {},
    width: 0,
    height: 0,
    theme: 'light',
    mode: 'view',
    tokens: {},
    error: '',
};

let animationFrame = 0;

function requestRender() {
    if (animationFrame) return;
    animationFrame = requestAnimationFrame(() => {
        animationFrame = 0;
        render();
    });
}
```

Coalesce listener updates into one animation frame. Data, options, dimensions, and theme can arrive separately during startup.

## Data normalization

```javascript
function normalizeSearchData(data) {
    const fields = (data?.fields || []).map((field) =>
        typeof field === 'string' ? field : field.name
    );
    const rows = data?.rows?.length
        ? data.rows
        : Array.from(
              { length: data?.columns?.[0]?.length || 0 },
              (_, rowIndex) => data.columns.map((column) => column[rowIndex])
          );
    const fieldIndex = Object.fromEntries(fields.map((name, index) => [name, index]));
    return { fields, rows, fieldIndex };
}

function toNumber(value, fallback = 0) {
    const parsed = Number.parseFloat(value);
    return Number.isFinite(parsed) ? parsed : fallback;
}
```

Convert the framework data into a small visualization model before drawing. Do not pass `dataSources` deep into rendering helpers.

## HiDPI Canvas setup

```javascript
function prepareCanvas(canvasElement, width, height) {
    if (!(width > 0 && height > 0)) return null;

    const ratio = Math.max(1, window.devicePixelRatio || 1);
    const pixelWidth = Math.round(width * ratio);
    const pixelHeight = Math.round(height * ratio);

    canvasElement.style.width = `${width}px`;
    canvasElement.style.height = `${height}px`;

    if (canvasElement.width !== pixelWidth || canvasElement.height !== pixelHeight) {
        canvasElement.width = pixelWidth;
        canvasElement.height = pixelHeight;
    }

    const context = canvasElement.getContext('2d');
    if (!context) return null;

    context.setTransform(ratio, 0, 0, ratio, 0, 0);
    context.clearRect(0, 0, width, height);
    return context;
}
```

Use the API-provided CSS dimensions for layout. Draw in CSS pixels after setting the transform; do not calculate geometry from the backing-store dimensions.

## Render state machine

```javascript
function drawMessage(context, width, height, message, theme) {
    context.save();
    context.fillStyle = theme === 'dark' ? '#ffffff' : '#3c444d';
    context.font = '14px sans-serif';
    context.textAlign = 'center';
    context.textBaseline = 'middle';
    context.fillText(message, width / 2, height / 2);
    context.restore();
}

function render() {
    const { width, height, loading, dataSources, options, theme } = state;
    const context = prepareCanvas(canvas, width, height);
    if (!context) return;

    if (loading) {
        drawMessage(context, width, height, 'Loading data…', theme);
        return;
    }

    const sourceData = dataSources?.primary?.data;
    if (!sourceData) {
        drawMessage(context, width, height, 'No data available', theme);
        return;
    }

    const normalized = normalizeSearchData(sourceData);
    if (normalized.rows.length === 0) {
        drawMessage(context, width, height, 'No data available', theme);
        return;
    }

    const valueField = options.valueField ?? 'count';
    const valueIndex = normalized.fieldIndex[valueField];
    if (valueIndex === undefined) {
        VisualizationAPI.setError(`Required field not found: ${valueField}`);
        return;
    }

    VisualizationAPI.clearError();
    const lastRow = normalized.rows[normalized.rows.length - 1];
    const model = {
        value: toNumber(lastRow[valueIndex]),
        color: options.valueColor ?? '#4e9cf5',
        maximum: toNumber(options.maximum, 100),
        theme,
    };

    drawVisualization(context, width, height, model);
}

function drawVisualization(context, width, height, model) {
    // Keep all visualization-specific Canvas work here or in imported pure helpers.
}
```

Use an in-frame message for ordinary loading and empty states. Use `setError` for an actionable invalid configuration or result shape. Always call `clearError` after valid data recovers.

## API subscriptions

```javascript
const cleanups = [
    VisualizationAPI.addDataSourcesListener((next) => {
        state.dataSources = next.dataSources || {};
        state.loading = Boolean(next.loading);
        requestRender();
    }, { invokeImmediately: true }),
    VisualizationAPI.addOptionsListener(({ options }) => {
        state.options = options || {};
        requestRender();
    }, { invokeImmediately: true }),
    VisualizationAPI.addDimensionsListener(({ width, height }) => {
        state.width = width || 0;
        state.height = height || 0;
        requestRender();
    }, { invokeImmediately: true }),
    VisualizationAPI.addThemeListener(({ theme }) => {
        state.theme = theme || 'light';
        requestRender();
    }, { invokeImmediately: true }),
    VisualizationAPI.addModeListener(({ mode }) => {
        state.mode = mode || 'view';
    }, { invokeImmediately: true }),
    VisualizationAPI.addTokensListener(({ tokens }) => {
        state.tokens = tokens || {};
    }, { invokeImmediately: true }),
];
```

Only subscribe to state the visualization uses. Store cleanup functions and expose a test/hot-reload teardown when useful:

```javascript
function destroy() {
    cleanups.forEach((cleanup) => cleanup?.());
    cancelAnimationFrame(animationFrame);
    animationFrame = 0;
    canvas.replaceWith();
}

if (import.meta.hot) {
    import.meta.hot.dispose(destroy);
}
```

## Canvas drilldown

Track hit regions in CSS pixels. A simple single-target visualization can attach the Dashboard Studio listener to the canvas:

```javascript
VisualizationAPI.addDrilldownListener({
    node: canvas,
    action: 'custom.click',
    payloadCallback: () => ({
        data: { field: 'host', value: currentModel.host },
        name: 'host',
        value: currentModel.host,
    }),
});
```

For per-mark drilldown, update the current payload from pointer coordinates before the click is handled, or install a click handler and call `triggerDrilldown({ action, payload, originalEvent })`. Keep `showDrilldown` and `hasEventHandlers` in `config.json` synchronized with the implementation.

## CSS baseline

```css
html,
body,
#root {
    width: 100%;
    height: 100%;
    margin: 0;
    overflow: hidden;
}

* {
    box-sizing: border-box;
}

```

Do not expect Dashboard Studio CSS variables, fonts, or inherited styles to cross the iframe boundary. Import or inline every required asset through the generated build pipeline.
